 * BACKEND RESMI ARAHH SWEET & SAVORY (Google Apps Script - Kode.gs)

var OWNER_EMAIL = "agniguntursugito@gmail.com";
var SPREADSHEET_NAME = "ARAHH Sweet & Savory — Rekapitulasi & Database Toko";

// 1. Dapatkan atau Buat File Google Spreadsheet Terpusat
function getOrCreateSpreadsheet() {
  try {
    var active = SpreadsheetApp.getActiveSpreadsheet();
    if (active) return active;
  } catch (e) {}

  var props = PropertiesService.getScriptProperties();
  var savedId = props.getProperty("ARAHH_SPREADSHEET_ID");
  if (savedId) {
    try {
      return SpreadsheetApp.openById(savedId);
    } catch (e) {}
  }

  var files = DriveApp.getFilesByName(SPREADSHEET_NAME);
  if (files.hasNext()) {
    var ss = SpreadsheetApp.open(files.next());
    props.setProperty("ARAHH_SPREADSHEET_ID", ss.getId());
    return ss;
  }

  var newSs = SpreadsheetApp.create(SPREADSHEET_NAME);
  props.setProperty("ARAHH_SPREADSHEET_ID", newSs.getId());
  return newSs;
}

// 2. Fungsi Helper Auto-Format Tabel Spreadsheet (Warna Header, Border, Zebra Banding, Currency, Lebar Kolom)
function autoFormatSheetTable(sheet, titleText, summaryText, headers, rows, currencyCols, colWidths) {
  sheet.clear();
  var totalCols = headers.length;

  // Baris 1: Judul Utama Sheet
  sheet.getRange(1, 1, 1, totalCols).merge()
    .setValue(titleText)
    .setBackground("#271714")
    .setFontColor("#FFF8E1")
    .setFontFamily("Plus Jakarta Sans")
    .setFontSize(12)
    .setFontWeight("bold")
    .setVerticalAlignment("middle")
    .setHorizontalAlignment("left");
  sheet.setRowHeight(1, 34);

  // Baris 2: Ringkasan Eksekutif & Waktu Sinkronisasi
  sheet.getRange(2, 1, 1, totalCols).merge()
    .setValue(summaryText)
    .setBackground("#FFF8E1")
    .setFontColor("#3E2723")
    .setFontSize(10)
    .setFontWeight("bold")
    .setVerticalAlignment("middle");
  sheet.setRowHeight(2, 26);

  // Baris 3: Header Tabel
  var headerRange = sheet.getRange(3, 1, 1, totalCols);
  headerRange.setValues([headers])
    .setBackground("#3E2723")
    .setFontColor("#FFFFFF")
    .setFontSize(10)
    .setFontWeight("bold")
    .setHorizontalAlignment("center")
    .setVerticalAlignment("middle")
    .setWrap(true);
  sheet.setRowHeight(3, 30);
  sheet.setFrozenRows(3);

  // Baris 4+: Isi Data
  if (rows && rows.length > 0) {
    var dataRange = sheet.getRange(4, 1, rows.length, totalCols);
    dataRange.setValues(rows)
      .setFontSize(10)
      .setVerticalAlignment("middle")
      .setWrap(true);

    // Zebra Striping & Border Rapi
    for (var r = 0; r < rows.length; r++) {
      var bg = (r % 2 === 0) ? "#FFFFFF" : "#FDF8F5";
      sheet.getRange(4 + r, 1, 1, totalCols).setBackground(bg);
      sheet.setRowHeight(4 + r, 26);
    }

    // Format Rupiah pada Kolom Harga / Tagihan
    if (currencyCols && currencyCols.length > 0) {
      for (var c = 0; c < currencyCols.length; c++) {
        var colIdx = currencyCols[c];
        sheet.getRange(4, colIdx, rows.length, 1)
          .setNumberFormat('"Rp"#,##0')
          .setHorizontalAlignment("right")
          .setFontWeight("bold");
      }
    }

    sheet.getRange(3, 1, rows.length + 1, totalCols)
      .setBorder(true, true, true, true, true, true, "#D7CCC8", SpreadsheetApp.BorderStyle.SOLID);
  } else {
    sheet.getRange(4, 1, 1, totalCols).merge()
      .setValue("Belum ada data tercatat.")
      .setHorizontalAlignment("center")
      .setFontStyle("italic")
      .setFontColor("#8D6E63");
  }

  // Atur Lebar Kolom Otomatis / Proporsional
  if (colWidths && colWidths.length === totalCols) {
    for (var i = 0; i < colWidths.length; i++) {
      sheet.setColumnWidth(i + 1, colWidths[i]);
    }
  } else {
    sheet.autoResizeColumns(1, totalCols);
  }
}

// 3. Sinkronisasi & Auto-Format 5 Sheet Rekapitulasi di Google Spreadsheet
function syncAndAutoFormatSpreadsheet(state) {
  var ss = getOrCreateSpreadsheet();
  var nowStr = Utilities.formatDate(new Date(), "GMT+7", "dd MMM yyyy HH:mm:ss") + " WIB";

  function ensureSheet(name) {
    var s = ss.getSheetByName(name);
    if (!s) s = ss.insertSheet(name);
    return s;
  }

  // A. SHEET 1: REKAP PESANAN PRE-ORDER
  var orders = state.ordersHistory || [];
  var totalOmzet = 0;
  var lunasCount = 0;
  var orderRows = orders.map(function(o, idx) {
    var total = Number(o.grandTotal || 0);
    totalOmzet += total;
    if (o.status === "Pembayaran Diterima" || o.status === "Selesai / Dikirim") lunasCount++;
    var itemsText = (o.items || []).map(function(it) {
      return it.qty + "x " + it.name + " (" + it.variantName + " | Topping: " + (it.toppings || []).join(", ") + ")" + (it.note ? " [" + it.note + "]" : "");
    }).join("\n");
    var proofStatus = o.paymentProof ? "Sudah Upload Bukti" : (String(o.paymentMethod || "").indexOf("Bayar Setelah") !== -1 ? "Pelanggan Dekat (Bayar Nanti)" : "Belum Upload");
    return [
      idx + 1,
      o.orderId || "-",
      o.createdAt || "-",
      o.buyerName || "-",
      "'" + (o.buyerPhone || "-"),
      o.buyerAddress || "-",
      o.batchDate || "-",
      itemsText || "-",
      o.paymentMethod || "-",
      o.status || "Menunggu Konfirmasi",
      proofStatus,
      total
    ];
  });

  var sOrders = ensureSheet("Rekap_Pesanan_PO");
  autoFormatSheetTable(
    sOrders,
    "📊 REKAPITULASI PESANAN PRE-ORDER — ARAHH SWEET & SAVORY",
    "Total Pesanan: " + orders.length + " Nota  |  Pesanan Lunas/Selesai: " + lunasCount + "  |  Total Omzet PO: Rp" + totalOmzet.toLocaleString("id-ID") + "  |  Update: " + nowStr,
    ["No", "No. Invoice", "Waktu Pesan", "Nama Pemesan", "No. WhatsApp", "Alamat Pengiriman", "Jadwal Batch PO", "Rincian Menu & Topping", "Metode Pembayaran", "Status Pesanan", "Bukti Bayar", "Total Tagihan (Rp)"],
    orderRows,
    [12],
    [45, 125, 130, 140, 120, 200, 140, 280, 210, 160, 150, 135]
  );

  // B. SHEET 2: KATALOG PRODUK & STOK KUOTA PO
  var products = state.products || [];
  var prodRows = products.map(function(p, idx) {
    var vars = (p.variants || []).map(function(v) { return v.name + " (" + (v.priceDelta >= 0 ? "+" : "") + v.priceDelta + ")"; }).join(", ");
    var tops = (p.toppings || []).map(function(t) { return t.name + " (+" + t.price + ")"; }).join(", ");
    return [
      idx + 1,
      p.id || "-",
      p.categoryName || p.category || "-",
      p.name || "-",
      Number(p.basePrice || 0),
      Number(p.oldPrice || 0),
      Number(p.quotaLeft || 0),
      Number(p.sold || 0),
      vars,
      tops,
      p.desc || "-"
    ];
  });

  var sProds = ensureSheet("Katalog_Produk");
  autoFormatSheetTable(
    sProds,
    "🍪 DATABASE KATALOG PRODUK, VARIAN & KUOTA PO — ARAHH SWEET & SAVORY",
    "Total Menu Aktif: " + products.length + " Produk  |  Terakhir Diperbarui: " + nowStr,
    ["No", "ID Produk", "Kategori", "Nama Produk", "Harga Dasar (Rp)", "Harga Coret (Rp)", "Sisa Kuota PO", "Terjual", "Varian Ukuran/Paket", "Varian Topping/Rasa", "Deskripsi"],
    prodRows,
    [5, 6],
    [45, 140, 130, 230, 125, 125, 100, 85, 240, 240, 250]
  );

  // C. SHEET 3: DAFTAR PELANGGAN DEKAT (BAYAR SETELAH BARANG DATANG)
  var known = state.knownCustomers || [];
  var knownRows = known.map(function(k, idx) {
    var stLabel = k.status === "approved" ? "DIKONFIRMASI KENAL (Bisa Bayar Nanti)" : (k.status === "pending" ? "MENUNGGU KONFIRMASI" : "DITOLAK / DICABUT");
    return [
      idx + 1,
      k.name || "-",
      "'" + (k.phone || "-"),
      k.relationNote || "-",
      stLabel,
      k.requestedAt || "-"
    ];
  });

  var sKnown = ensureSheet("Pelanggan_Dekat");
  autoFormatSheetTable(
    sKnown,
    "🤝 DAFTAR VERIFIKASI PELANGGAN DEKAT (IZIN BAYAR SETELAH BARANG DATANG)",
    "Total Terdaftar: " + known.length + " Pelanggan  |  Update: " + nowStr,
    ["No", "Nama Pelanggan", "No. WhatsApp", "Identitas / Keterangan Kenalan", "Status Verifikasi Admin", "Waktu Pengajuan"],
    knownRows,
    [],
    [45, 180, 135, 260, 230, 140]
  );

  // D. SHEET 4: METODE PEMBAYARAN & IDENTITAS TOKO
  var cfg = state.storeConfig || {};
  var pays = state.paymentMethods || [];
  var payRows = pays.map(function(pm, idx) {
    return [
      idx + 1,
      idx === 0 ? "UTAMA" : "TAMBAHAN",
      pm.bankName || "-",
      "'" + (pm.accNumber || "-"),
      pm.accHolder || "-",
      (cfg.storeName || "ARAHH") + " " + (cfg.storeSubtitle || "Sweet & Savory"),
      (cfg.waAdmin1Name || "Guntur") + " (" + (cfg.waNumber || "-") + ") / " + (cfg.waAdmin2Name || "Rini") + " (" + (cfg.waNumber2 || "-") + ")"
    ];
  });

  var sPay = ensureSheet("Metode_Pembayaran_&_Toko");
  autoFormatSheetTable(
    sPay,
    "🏦 METODE PEMBAYARAN & IDENTITAS PENGELOLA — ARAHH SWEET & SAVORY",
    "Brand: " + (cfg.storeName || "ARAHH") + "  |  Jadwal: " + (cfg.batchInfo || "-") + "  |  Update: " + nowStr,
    ["No", "Prioritas", "Nama Bank / E-Wallet", "Nomor Rekening / Tujuan", "Atas Nama & Keterangan", "Identitas Brand Toko", "Kontak Admin WhatsApp"],
    payRows,
    [],
    [45, 100, 180, 150, 250, 180, 250]
  );

  // E. SHEET 5: TESTIMONI & RATING PELANGGAN
  var revs = state.reviews || [];
  var revRows = revs.map(function(rv, idx) {
    return [
      idx + 1,
      rv.date || "-",
      rv.customerName || "-",
      rv.productName || "-",
      Number(rv.rating || 5) + " ★",
      rv.comment || "-"
    ];
  });

  var sRev = ensureSheet("Testimoni_Pelanggan");
  autoFormatSheetTable(
    sRev,
    "⭐ REKAP TESTIMONI & RATING PEMBELI — ARAHH SWEET & SAVORY",
    "Total Testimoni Ditampilkan: " + revs.length + " Ulasan  |  Update: " + nowStr,
    ["No", "Tanggal", "Nama Pelanggan", "Menu yang Dipesan", "Rating Bintang", "Ulasan / Testimoni"],
    revRows,
    [],
    [45, 120, 160, 220, 110, 320]
  );

  // Hapus Sheet1 default kosong jika masih ada
  var defaultSheet = ss.getSheetByName("Sheet1");
  if (defaultSheet && ss.getSheets().length > 1) {
    try { ss.deleteSheet(defaultSheet); } catch (e) {}
  }

  return ss.getUrl();
}

function getOrCreateDbFile() {
  var files = DriveApp.getFilesByName(DB_FILE_NAME);
  if (files.hasNext()) return files.next();
  var initialData = { updatedAt: new Date().toISOString() };
  return DriveApp.createFile(DB_FILE_NAME, JSON.stringify(initialData), MimeType.PLAIN_TEXT);
}

function readDbState() {
  try {
    var file = getOrCreateDbFile();
    var raw = file.getBlob().getDataAsString();
    return raw ? JSON.parse(raw) : {};
  } catch (err) {
    return {};
  }
}

function writeDbState(data) {
  var file = getOrCreateDbFile();
  data.updatedAt = new Date().toISOString();
  file.setContent(JSON.stringify(data));
  try {
    data.spreadsheetUrl = syncAndAutoFormatSpreadsheet(data);
  } catch (sheetErr) {}
  return data;
}

function doGet(e) {
  var state = readDbState();
  try {
    if (!state.spreadsheetUrl) {
      state.spreadsheetUrl = getOrCreateSpreadsheet().getUrl();
    }
  } catch (err) {}
  return ContentService.createTextOutput(JSON.stringify({
    status: "ok",
    spreadsheetUrl: state.spreadsheetUrl || "",
    data: state
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents || "{}");
    var action = body.action || "syncAll";

    if (action === "sendOtp") {
      var otp = body.otp || "882491";
      MailApp.sendEmail({
        to: OWNER_EMAIL,
        subject: "[ARAHH Admin] Kode Verifikasi OTP Login: " + otp,
        body: "Halo Pengelola ARAHH Sweet & Savory,\n\nKode OTP verifikasi login Portal Pengelola Anda adalah: " + otp
      });
      return ContentService.createTextOutput(JSON.stringify({ status: "ok", message: "OTP terkirim" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var current = readDbState();
    var payload = body.payload || {};
    if (payload.storeConfig) current.storeConfig = payload.storeConfig;
    if (payload.paymentMethods) current.paymentMethods = payload.paymentMethods;
    if (payload.products) current.products = payload.products;
    if (payload.reviews) current.reviews = payload.reviews;
    if (payload.ordersHistory) current.ordersHistory = payload.ordersHistory;
    if (payload.knownCustomers) current.knownCustomers = payload.knownCustomers;

    var saved = writeDbState(current);
    return ContentService.createTextOutput(JSON.stringify({
      status: "ok",
      updatedAt: saved.updatedAt,
      spreadsheetUrl: saved.spreadsheetUrl || "",
      data: saved
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
PANDUAN MEMILIH DATABASE

Apa Bedanya Data di Google Spreadsheet (Apps Script) vs. Data di Firebase?

Website ARAHH Sweet & Savory ini mendukung kedua jenis database tersebut. Berikut perbedaan lengkapnya agar Anda bisa memilih sesuai kebutuhan:

Aspek Perbandingan Google Spreadsheet / Drive (via Apps Script) Firebase Realtime Database (Google Cloud)
Bentuk Penyimpanan Data Berbentuk Tabel Baris & Kolom (Sheet) / File di Google Drive milik akun Google Anda sendiri. Sangat mudah dibuka dan direkap di Excel. Berbentuk Pohon Data JSON (NoSQL) berkecepatan tinggi di server Google Cloud (console.firebase.google.com).
Kecepatan Sinkronisasi (HP ↔ Laptop) Cepat (~1 hingga 2 detik per request). Sangat memadai untuk sistem Pre-Order (PO) harian/mingguan. Instan / Real-time (< 0,2 detik). Begitu Admin klik di Laptop, tampilan di HP berubah detik itu juga.
Fitur Kirim Email OTP Login Bisa langsung kirim Email OTP gratis lewat Gmail Anda menggunakan MailApp.sendEmail. Khusus penyimpanan data cepat (tidak mengirim email kustom kecuali ditambah layanan pihak ketiga).
Kapasitas Foto Produk & Bukti Bayar Sangat lega karena tersimpan di kuota Google Drive (15 GB gratis). Kuota gratis Spark Plan 1 GB penyimpanan total & 10 GB transfer/bulan (foto otomatis dikompresi di web ini agar hemat).
Kemudahan Rekap Pembukuan Unggul untuk laporan keuangan manual karena menyatu dengan ekosistem Google Drive & Sheets. Unggul untuk performa aplikasi skala besar dengan ratusan pengguna bersamaan.
